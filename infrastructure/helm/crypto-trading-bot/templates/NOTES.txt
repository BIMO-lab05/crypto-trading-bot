========================================================================================
  🔧 GETTING STARTED
========================================================================================

1. Check deployment status:
   kubectl get pods -n {{ .Values.global.namespace }}

2. View service logs:
   kubectl logs -f -n {{ .Values.global.namespace }} -l app=api-gateway

3. Access API Gateway:
   {{- if .Values.api-gateway.ingress.enabled }}
   curl https://{{ (index .Values.api-gateway.ingress.hosts 0).host }}/health
   {{- else }}
   kubectl port-forward -n {{ .Values.global.namespace }} svc/api-gateway {{ .Values.api-gateway.service.port }}:{{ .Values.api-gateway.service.port }}
   curl http://localhost:{{ .Values.api-gateway.service.port }}/health
   {{- end }}

========================================================================================
  🛠️  REQUIRED MANUAL STEPS
========================================================================================

1. Create required secrets (if not already created):

   # PostgreSQL
   kubectl create secret generic postgresql-secret \
     --from-literal=password=YOUR_POSTGRES_PASSWORD \
     -n {{ .Values.global.namespace }}

   # TimescaleDB
   kubectl create secret generic timescaledb-secret \
     --from-literal=password=YOUR_TIMESCALE_PASSWORD \
     -n {{ .Values.global.namespace }}

   # Redis
   kubectl create secret generic redis-secret \
     --from-literal=password=YOUR_REDIS_PASSWORD \
     -n {{ .Values.global.namespace }}

   # RabbitMQ
   kubectl create secret generic rabbitmq-secret \
     --from-literal=password=YOUR_RABBITMQ_PASSWORD \
     -n {{ .Values.global.namespace }}

   # Bybit API
   kubectl create secret generic bybit-secret \
     --from-literal=api-key=YOUR_BYBIT_API_KEY \
     --from-literal=api-secret=YOUR_BYBIT_API_SECRET \
     -n {{ .Values.global.namespace }}

   # Grafana Admin
   kubectl create secret generic grafana-secret \
     --from-literal=password=YOUR_GRAFANA_PASSWORD \
     -n {{ .Values.global.namespace }}

2. Verify all pods are running:
   kubectl get pods -n {{ .Values.global.namespace }} --watch

3. Check service health:
   kubectl get svc -n {{ .Values.global.namespace }}
