```python
"""Mtime-watching model loader for INFRA-06 Bug 1.

Singleton pattern: get_current_model() returns the loaded model, reloading from disk if
the file's mtime changed since last load. Solves the "stale in-memory model after retrain"
bug — operator no longer needs to restart ml-prediction-service after running training.
"""
import logging
import threading
from pathlib import Path
from typing import Any, Optional

logger = logging.getLogger(__name__)

_LOAD_LOCK = threading.Lock()


class ModelLoader:
    def __init__(self, model_path: Path, loader_fn):
        self.model_path = Path(model_path)
        self.loader_fn = loader_fn  # callable: Path -> model object
        self._model = None
        self._mtime = 0.0

    def get_current_model(self) -> Any:
        if not self.model_path.exists():
            raise FileNotFoundError(f"Model file missing: {self.model_path}")
        current_mtime = self.model_path.stat().st_mtime
        if self._model is not None and current_mtime == self._mtime:
            return self._model
        with _LOAD_LOCK:
            # Re-check inside lock to avoid double-load
            current_mtime = self.model_path.stat().st_mtime
            if self._model is not None and current_mtime == self._mtime:
                return self._model
            logger.warning(
                "MODEL_RELOAD: path=%s mtime=%s (was=%s)",
                self.model_path, current_mtime, self._mtime,
            )
            self._model = self.loader_fn(self.model_path)
            self._mtime = current_mtime
        return self._model
```

```python
from pathlib import Path
from typing import Optional

import pytest


def _resolve_compose_value(service_block: str, pattern: str) -> Optional[str]:
    """Read docker-compose.unified.yml; extract a field from the named service block."""
    repo_root = Path(__file__).resolve().parents[2]
    compose = (repo_root / "docker-compose.unified.yml").read_text()
    # Capture from `^  <service>:` to the next sibling service header.
    match = re.search(
        rf"^  {re.escape(service_block)}:\n(.*?)(?=^  [a-z][a-z-]*:$)",
        compose, re.DOTALL | re.MULTILINE,
    )
    if not match:
        return None
    block = match.group(1)
    m = re.search(pattern, block)
    return m.group(1).strip() if m else None


def _resolve_ml_container_name() -> str:
    name = _resolve_compose_value("ml-prediction", r"container_name:\s*(\S+)")
    if not name:
        pytest.fail(
            "Could not resolve ml-prediction container_name from docker-compose.unified.yml. "
            "Update the compose block or rerun with the correct service name."
        )
    return name


def _resolve_first_model_file_inside_container() -> str:
    """List model files on host, pick the first one, and form its in-container path."""
    repo_root = Path(__file__).resolve().parents[2]
    host_dir = repo_root / "services" / "ml-prediction-service" / "models"
    candidates = sorted(
        list(host_dir.glob("*.h5")) + list(host_dir.glob("*.keras")) + list(host_dir.glob("*.pkl"))
    )
    if not candidates:
        pytest.fail(
            f"No model files in {host_dir}. Train a model first or skip this regression "
            "by deferring INFRA-06 Bug 1 in the triage gate (Task 1)."
        )
    return f"/app/models/{candidates[0].name}"


@pytest.mark.asyncio
async def test_stale_ml_model_reload(bootstrap_stack, services_config, http_client):
    """Bug 1 regression: touching the model file forces ml-prediction-service to reload.
    Container name + model path resolved from docker-compose.unified.yml (Blocker #4)."""
    container = _resolve_ml_container_name()
    model_path_inside_container = _resolve_first_model_file_inside_container()

    # Touch the model file to bump mtime (simulates retrain).
    res = subprocess.run(
        ["docker", "exec", container, "touch", model_path_inside_container],
        capture_output=True, text=True
    )
    if res.returncode != 0:
        pytest.fail(
            f"Could not touch {model_path_inside_container} inside container {container!r}: "
            f"stderr={res.stderr!r}. INFRA-06 Bug 1 fix may need adjustment, OR the resolved "
            f"container/path are wrong (verify docker-compose.unified.yml ml-prediction block)."
        )

    # Hit prediction endpoint twice; the second call should have triggered reload.
    url = f"{services_config['ml_prediction']}/api/v1/predictions/SOLUSDT"
    r1 = await http_client.get(url, timeout=10.0)
    assert r1.status_code == 200
    r2 = await http_client.get(url, timeout=10.0)
    assert r2.status_code == 200

    # Assert the reload log line appeared.
    logs = subprocess.run(
        ["docker", "logs", "--tail", "100", container],
        capture_output=True, text=True
    ).stdout
    assert "MODEL_RELOAD: path=" in logs, (
        f"Expected MODEL_RELOAD log line not found in {container!r} logs. "
        "Verify Plan 02-08 Task 2 wired the ModelLoader into the prediction path."
    )
```
