        return True
    except Exception as e:
        logger.error(f"Failed to send Telegram message: {e}")

"""CD-01 notification-delivery tests.

Two paths covered:
  - record mode (local default): notification-service writes JSON lines to tests/.notifications.log
  - live mode  (CI, opt-in):     notification-service POSTs to api.telegram.org; assert via getUpdates
"""
import os
import re
import time

import httpx
import pytest


# Notification-service test endpoint — find the existing route in services/notification-service/app/main.py
# Most likely: POST /notify/test or POST /notify with a body. Executor confirms when reading main.py.
NOTIFY_TEST_PATH = "/notify/test"


@pytest.mark.asyncio
async def test_notification_record_mode(
    bootstrap_stack, services_config, http_client, notification_log
):
    """Record mode default: notification writes JSON line to tests/.notifications.log."""
    if os.getenv("NOTIFICATION_TEST_MODE", "") not in ("", "record"):
        pytest.fail(
            "Test environment is not in record mode; set NOTIFICATION_TEST_MODE=record "
            "(or unset) and re-run, or run the live-mode test instead. D-12 forbids skip."
        )
    msg = f"phase2-record-test-{int(time.time())}"
    url = f"{services_config['notification']}{NOTIFY_TEST_PATH}"
    r = await http_client.post(url, json={"message": msg, "channel": "telegram"})
    assert r.status_code == 200, f"notify endpoint failed: {r.status_code} {r.text}"

    # Allow short flush window
    deadline = time.monotonic() + 5.0
    lines = []
    while time.monotonic() < deadline:
        lines = notification_log()
        if any(msg in ln for ln in lines):
            break
        time.sleep(0.25)
    assert any(msg in ln for ln in lines), (
        f"Notification not recorded in tests/.notifications.log. Got {len(lines)} lines."
    )


@pytest.mark.asyncio
async def test_notification_live_mode(
    bootstrap_stack, services_config, http_client
):
    """Live mode (CI opt-in): notification-service POSTs to api.telegram.org; verify via getUpdates."""
    bot_token = os.getenv("TEST_TELEGRAM_BOT_TOKEN", "")
    chat_id = os.getenv("TEST_TELEGRAM_CHAT_ID", "")
    if not (bot_token and chat_id and os.getenv("NOTIFICATION_TEST_MODE") == "live"):
        # NOT a skip — operator opted into record-mode by default. Live test is the alternate code path,
        # exercised only when the env vars are present (CI). When they're absent, the test is correctly
        # not applicable; bailing here is consistent with the local-vs-CI split per D-07.
        pytest.fail(
            "Live mode requires TEST_TELEGRAM_BOT_TOKEN, TEST_TELEGRAM_CHAT_ID, "
            "and NOTIFICATION_TEST_MODE=live. Run record-mode test in local dev."
        )
    msg = f"phase2-live-test-{int(time.time())}"
    url = f"{services_config['notification']}{NOTIFY_TEST_PATH}"
    r = await http_client.post(url, json={"message": msg, "channel": "telegram"})
    assert r.status_code == 200

    # getUpdates: poll telegram for incoming messages bot received
    deadline = time.monotonic() + 15.0
    found = False
    while time.monotonic() < deadline:
        async with httpx.AsyncClient() as client:
            resp = await client.get(
                f"https://api.telegram.org/bot{bot_token}/getUpdates",
                timeout=10.0,
            )
        if resp.status_code == 200:
            data = resp.json()
            for upd in data.get("result", []):
                m = upd.get("message", {}).get("text", "")
                if msg in m:
                    found = True
                    break
        if found:
            break
        time.sleep(1.0)
    assert found, f"Live-mode message not received via getUpdates within 15s: {msg}"
