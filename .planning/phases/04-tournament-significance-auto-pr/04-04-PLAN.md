<objective>
Implement the `tournament reproduce {tid} --git-sha {sha}` CLI subcommand that proves a winning tournament's significance is bit-(near-)identical when re-run from the snapshot at the recorded git_sha.

Purpose: Phase 3 staked everything on `(config + git_sha + tournament_start_ts) → identical leaderboard`. Phase 4 extends that to `(snapshot + git_sha + bootstrap_seed) → identical significance`. The `reproduce` subcommand is the executable proof. The idempotency CI test (`test_reproduce_idempotent.py`) is the load-bearing regression check — if it ever fails, reproducibility is broken and the operator must investigate.

Output: A non-interactive reproducer that exits 0 only when the diff is within `Sharpe diff < 1e-6, p-value diff < 0.005` (D-12 FP-noise tolerance). Exits 3 on dirty tree or HEAD mismatch, exits 4 on reproducibility break.
</objective>

    ```python
    from __future__ import annotations
    import json, logging, sys
    from pathlib import Path
    from typing import Any, Dict

    from app.orchestrator.launcher import _git_sha, _git_is_dirty
    from app.leaderboard.db import LeaderboardDB, run_migrations

    logger = logging.getLogger(__name__)

    HARNESS_ROOT = Path(__file__).resolve().parents[2]
    SHARPE_TOLERANCE = 1e-6      # D-12
    PVALUE_TOLERANCE = 0.005     # D-12

    def _validate_tid(tid: str) -> None:
        if not tid or any(ch in tid for ch in ("/", "..", "\0", " ")):
            raise ValueError(f"invalid tournament_id: {tid!r}")

    def _temp_db_path(tournament_id: str) -> Path:
        _validate_tid(tournament_id)
        return HARNESS_ROOT / "data" / "leaderboard" / f"reproduce_{tournament_id}.db"

    def _diff_significance(original: Dict[str, Any], reproduced: Dict[str, Any]) -> list[str]:
        """Return list of diff strings; empty list == within tolerance."""
        diffs = []
        for sym, orig in original.get("per_symbol", {}).items():
            rep = reproduced.get("per_symbol", {}).get(sym, {})
            for key, tol in (("sharpe_lift", SHARPE_TOLERANCE),
                             ("dir_acc_lift", SHARPE_TOLERANCE),
                             ("sharpe_pvalue", PVALUE_TOLERANCE),
                             ("dir_acc_pvalue", PVALUE_TOLERANCE)):
                a = float(orig.get(key, 0.0))
                b = float(rep.get(key, 0.0))
                if abs(a - b) > tol:
                    diffs.append(f"{sym}.{key}: {a} vs {b} (delta={abs(a - b):.6g}, tol={tol})")
        return diffs

    def run_reproduce(
        tournament_id: str,
        *,
        git_sha_expected: str,
        force: bool = False,
    ) -> int:
        _validate_tid(tournament_id)

        # 1. Dirty tree refusal — D-12: no --allow-dirty here.
        if _git_is_dirty():
            sys.stderr.write(
                "git tree dirty - `reproduce` REFUSES (no --allow-dirty here; "
                "reproducibility is the whole point).\n"
            )
            raise SystemExit(3)
        head = _git_sha()
        if head != git_sha_expected:
            sys.stderr.write(
                f"HEAD ({head}) != requested git_sha ({git_sha_expected}). "
                f"Run: git checkout {git_sha_expected}\n"
            )
            raise SystemExit(3)

        # 2. Snapshot + original significance must exist.
        snap_dir = HARNESS_ROOT / "data" / "snapshots"
        snapshot_path = snap_dir / f"{tournament_id}.json"
        if not snapshot_path.exists():
            sys.stderr.write(f"snapshot not found: {snapshot_path}\n")
            raise SystemExit(2)
        with open(snapshot_path) as f:
            snapshot = json.load(f)
        original_sig_path = snap_dir / f"{tournament_id}.significance.json"
        if not original_sig_path.exists():
            sys.stderr.write(
                f"original significance not found at {original_sig_path}; "
                "run `tournament open-pr` first to record a baseline.\n"
            )
            raise SystemExit(2)
        with open(original_sig_path) as f:
            original_sig = json.load(f)

        # 3. Temp DB rebuild from snapshot rows (CD-06).
        temp_db = _temp_db_path(tournament_id)
        if temp_db.exists():
            if not force:
                sys.stderr.write(
                    f"temp DB exists at {temp_db}; pass --force to overwrite, "
                    "or delete it manually for forensic inspection.\n"
                )
                raise SystemExit(2)
            temp_db.unlink()
        temp_db.parent.mkdir(parents=True, exist_ok=True)
        run_migrations(temp_db)
        db = LeaderboardDB(temp_db)
        try:
            db.upsert_tournament(
                snapshot["tournament_id"],
                snapshot["config"]["config_yaml"],
                snapshot["config"].get("git_sha", "unknown"),
                int(snapshot["config"].get("seed", 0)),
            )
            for row in snapshot["rows"]:
                db.insert_run(row)
        finally:
            db.close()

        # 4. Re-run open-pr pipeline in dry-run mode using a `dry-run` output suffix so the
        #    operator's original {tid}.significance.json (and ensemble.json + leaderboard.md) are
        #    NEVER overwritten. W2 fix: drop the backup/restore choreography that was non-atomic
        #    and risked losing the original on crash between `write_text` calls.
        from app.pr.open_pr import run_open_pr   # lazy import (heavy)
        DRYRUN_SUFFIX = "dry-run"
        rc = run_open_pr(tournament_id, allow_dirty=False, dry_run=True, output_suffix=DRYRUN_SUFFIX)
        if rc != 0:
            sys.stderr.write(f"open-pr dry-run returned non-zero rc={rc}\n")
            raise SystemExit(rc)
        # Reproduced artifacts land at {tid}.{dry-run}.significance.json etc.
        reproduced_sig_path = snap_dir / f"{tournament_id}.{DRYRUN_SUFFIX}.significance.json"
        if not reproduced_sig_path.exists():
            sys.stderr.write(
                f"open-pr dry-run did not write expected artifact: {reproduced_sig_path}\n"
            )
            raise SystemExit(4)
        with open(reproduced_sig_path) as f:
            reproduced_sig = json.load(f)
        diffs = _diff_significance(original_sig, reproduced_sig)
        if diffs:
            sys.stderr.write("reproducibility broken — diffs:\n  " + "\n  ".join(diffs) + "\n")
            # CD-06: leave temp DB AND dry-run artifacts in place for forensic inspection.
            raise SystemExit(4)
        # Success: original artifacts untouched (mtime preserved); clean up temp DB and dry-run scratch.
        temp_db.unlink(missing_ok=True)
        for ext in ("significance.json", "ensemble.json", "leaderboard.md"):
            scratch = snap_dir / f"{tournament_id}.{DRYRUN_SUFFIX}.{ext}"
            scratch.unlink(missing_ok=True)
        print(json.dumps({"tournament_id": tournament_id, "git_sha": head, "diffs": []}, indent=2))
        return 0
    ```
