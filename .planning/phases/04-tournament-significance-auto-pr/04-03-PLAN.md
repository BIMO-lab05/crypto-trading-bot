    ```python
    from __future__ import annotations
    from typing import Any, Dict, Iterable, List
    from datetime import datetime, timezone

    MAX_BODY_CHARS = 60_000   # CD-12

    def render_pr_title(*, tournament_id: str, n_winning_symbols: int, n_total_symbols: int) -> str:
        """CD-01: ≤90 chars; truncate tournament_id to 12 chars if needed."""
        short = tournament_id[:12]
        return f"Tournament {short}: ensemble wins {n_winning_symbols}/{n_total_symbols} symbols (p<0.05 vs persistence)"

    def render_leaderboard_markdown(snapshot: Dict[str, Any], significance: Dict[str, Any], ensembles: Dict[str, List[Dict]]) -> str:
        """CD-05: top-5 per symbol, GFM table; rows in selected ensembles marked with ★; symbol status tag."""
        lines = []
        for symbol in snapshot["summary"]["symbols"]:
            sig = significance["per_symbol"].get(symbol, {})
            tag = (
                "[WIN]" if sig.get("win_gate_passed")
                else "[insufficient runs]" if "insufficient_runs" in (sig.get("gate_failure_reasons") or [])
                else "[no win]"
            )
            lines.append(f"### {symbol} {tag}")
            lines.append("")
            lines.append("| ★ | architecture | hp_hash | dsr | psr | cpcv_dsr | oos_sharpe | dir_acc_corrected |")
            lines.append("|---|--------------|---------|-----|-----|----------|------------|-------------------|")
            ensemble_run_ids = {m["run_id"] for m in ensembles.get(symbol, [])}
            top_5 = sorted(
                [r for r in snapshot["rows"] if r["symbol"] == symbol and r.get("status") == "success"],
                key=lambda r: -float(r.get("dsr") or 0.0),
            )[:5]
            for r in top_5:
                star = "★" if r["run_id"] in ensemble_run_ids else " "
                lines.append(
                    f"| {star} | {r['architecture']} | {r['hp_hash']} | "
                    f"{float(r.get('dsr') or 0.0):.4f} | {float(r.get('psr') or 0.0):.4f} | "
                    f"{float(r.get('cpcv_dsr') or 0.0):.4f} | {float(r.get('oos_sharpe') or 0.0):.4f} | "
                    f"{float(r.get('dir_acc_corrected') or 0.0):.4f} |"
                )
            lines.append("")
        return "\n".join(lines)

    def _render_disclosure(*, tournaments_evaluated_count: int, prior_tids: Iterable[str]) -> str:
        priors = list(prior_tids)[:20]
        bonferroni = (
            "## Disclosure (anti-cherry-picking, D-09)\n\n"
            f"- tournaments_evaluated_count: {tournaments_evaluated_count}\n"
            "- p_adjusted ≈ p_value × tournaments_evaluated_count\n"
            "- DSR deflates within a tournament, NOT across tournaments — multiple-testing correction is the operator's job.\n"
            f"- Prior tournament_ids (most recent {len(priors)}): "
            + (", ".join(f"`{t}`" for t in priors) if priors else "_(none)_")
        )
        return bonferroni

    def render_pr_body(
        *,
        tournament_id: str,
        git_sha: str,
        snapshot: Dict[str, Any],
        ensembles: Dict[str, List[Dict[str, Any]]],
        significance: Dict[str, Any],
        leaderboard_markdown: str,
        leaderboard_md_relative_path: str,         # for CD-12 fallback link
        tournaments_evaluated_count: int,
        prior_tournament_ids: Iterable[str],
    ) -> str:
        """Returns the full PR body. Falls back to summary+link if > MAX_BODY_CHARS (CD-12)."""
        n_wins = significance.get("n_winning_symbols", 0)
        n_total = len(snapshot["summary"]["symbols"])
        no_win = "true" if n_wins == 0 else "false"
        header = (
            f"# Tournament Significance — {tournament_id}\n\n"
            f"- git_sha: `{git_sha}`\n"
            f"- evaluated_at: {datetime.now(timezone.utc).isoformat()}\n"
            f"- n_winning_symbols: {n_wins} / {n_total}\n"
            f"- no_win: {no_win}\n"
            f"- baseline: persistence\n"
            f"- aggregation: mean_log_returns\n"
        )
        ensemble_md = "## Ensemble (config-by-reference)\n\n```json\n" + _short_json(ensembles) + "\n```"
        sig_md = "## Significance\n\n```json\n" + _short_json(significance) + "\n```"
        repro_md = (
            "## Reproduce\n\n```bash\n"
            f"python -m services.tournament_harness.app.cli reproduce {tournament_id} --git-sha {git_sha}\n"
            "```\n"
        )
        lb_md = "## Leaderboard\n\n" + leaderboard_markdown
        disc_md = _render_disclosure(tournaments_evaluated_count=tournaments_evaluated_count,
                                     prior_tids=prior_tournament_ids)
        full = "\n\n".join([header, lb_md, ensemble_md, sig_md, repro_md, disc_md])
        if len(full) <= MAX_BODY_CHARS:
            return full
        # CD-12: fallback summary
        link = f"\n\n_Leaderboard exceeded {MAX_BODY_CHARS} chars; full table at_ `{leaderboard_md_relative_path}`"
        summary = "\n\n".join([header, ensemble_md, sig_md, repro_md, disc_md]) + link
        # If even the summary exceeds the cap, drop sig/ensemble JSON to a count + link.
        if len(summary) > MAX_BODY_CHARS:
            summary = (header + "\n\n" + repro_md + "\n\n" + disc_md +
                       f"\n\n_Body truncated; see `{leaderboard_md_relative_path}` and the artifacts in the PR diff._")
        return summary

    def _short_json(obj) -> str:
        import json
        return json.dumps(obj, indent=2, default=str, sort_keys=True)
    ```

    ```python
    from __future__ import annotations
    import logging, os, re, subprocess, sys
    from typing import Sequence

    logger = logging.getLogger(__name__)

    _BRANCH_RE = re.compile(r"^[A-Za-z0-9._\-/]+$")

    def _check_gh_installed() -> None:
        """CD-09: hard-fail with exit 2 if gh missing."""
        try:
            subprocess.check_output(["gh", "--version"], stderr=subprocess.STDOUT)
        except (FileNotFoundError, subprocess.CalledProcessError) as e:
            sys.stderr.write(
                f"gh CLI not found on PATH. Install: https://cli.github.com/  (underlying error: {e})\n"
            )
            raise SystemExit(2)

    def _check_gh_token() -> None:
        """D-11: hard-fail with exit 2 if GH_TOKEN unset."""
        if not os.environ.get("GH_TOKEN"):
            sys.stderr.write(
                "GH_TOKEN env var not set — refusing to invoke gh pr create. "
                "(D-11: missing PR after winning tournament is the worst silent failure.)\n"
            )
            raise SystemExit(2)

    def open_draft_pr(
        *,
        title: str,
        body: str,
        head: str,                                  # CD-02 branch
        base: str = "main",
        labels: Sequence[str] = ("tournament", "evaluation-gate"),
        dry_run: bool = False,
    ) -> str:
        """Returns PR URL on success; '(dry-run)' if dry_run=True. NEVER invokes `gh pr merge` (CD-10)."""
        if not _BRANCH_RE.fullmatch(head):
            raise ValueError(f"invalid branch name {head!r} — must match {_BRANCH_RE.pattern}")
        cmd = [
            "gh", "pr", "create", "--draft",
            "--title", title,
            "--body", body,
            "--head", head,
            "--base", base,
        ]
        for label in labels:
            cmd += ["--label", label]
        if dry_run:
            # Log the cmd shape but NOT GH_TOKEN. cmd never contains the token.
            logger.info("dry-run gh pr create: argv=%s", cmd)
            return "(dry-run)"
        # Belt-and-suspenders: env hygiene.
        _check_gh_installed()
        _check_gh_token()
        # subprocess inherits env (gh reads GH_TOKEN from env). Do NOT pass a custom env that
        # could log the token via str(env) anywhere downstream.
        result = subprocess.run(cmd, capture_output=True, text=True, check=False, shell=False)
        if result.returncode != 0:

    ```python
    def dir_acc_corrected_from_log_returns(
        actual_lr: np.ndarray,
        pred_lr: np.ndarray,
    ) -> float:
        """Chance-corrected directional accuracy on log-return series (D-02).

        D-02 mandates ensemble metrics are computed directly on the averaged
        log-return series — no re-conversion to price for metric purposes.
        compute_returns_metrics consumes price arrays, so this is the canonical
        log-return-input sibling. Equivalent semantics: a model with no skill
        scores ~0; perfect sign agreement scores 1.0; perfect sign disagreement
        scores -1.0.

        Returns 2 * (mean(sign(actual_lr) == sign(pred_lr)) - 0.5).

        NOTE: when pred_lr is the persistence baseline (all zeros), np.sign(0)
        is 0 and never agrees with the non-zero actual signs, so the function
        returns -1.0. This is the INTENDED chance baseline — persistence has
        no directional skill on log-returns by construction.
        """
        actual_lr = np.asarray(actual_lr, dtype=float)
        pred_lr = np.asarray(pred_lr, dtype=float)
        if actual_lr.shape != pred_lr.shape:
            raise ValueError(
                f"shape mismatch: actual_lr={actual_lr.shape} pred_lr={pred_lr.shape}"
            )
        if actual_lr.size == 0:
            return float("nan")
        agree = (np.sign(actual_lr) == np.sign(pred_lr)).astype(float)
        return 2.0 * (float(np.mean(agree)) - 0.5)
    ```
