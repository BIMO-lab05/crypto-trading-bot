    ```python
    from __future__ import annotations
    import hashlib, math
    from typing import Callable, Dict, Literal
    import numpy as np

    Alternative = Literal["greater"]

    def derive_seed(tournament_id: str, symbol: str) -> int:
        """CD-07: tournament-id + symbol derived 31-bit seed."""
        digest = hashlib.blake2b(f"{tournament_id}|sig|{symbol}".encode(), digest_size=8).hexdigest()
        return int(digest, 16) & 0x7FFFFFFF

    def _generate_block_resample(rng: np.random.Generator, n: int, block_size: int) -> np.ndarray:
        """Politis-Romano stationary block: each block has geometric length with mean = block_size."""
        idxs = np.empty(n, dtype=np.int64)
        i = 0
        p = 1.0 / block_size
        while i < n:
            start = int(rng.integers(0, n))
            # Geometric block length with mean = block_size; clamp to remaining slots.
            length = int(rng.geometric(p))
            length = min(length, n - i)
            for k in range(length):
                idxs[i + k] = (start + k) % n   # circular wrap (Politis-Romano)
            i += length
        return idxs

    def stationary_block_bootstrap_pvalue(
        paired_diffs: np.ndarray,                   # d_t = ensemble_log_ret_t - baseline_log_ret_t (= ensemble_log_ret_t for persistence)
        *,
        metric_fn: Callable[[np.ndarray], float],   # e.g. lambda r: r.mean() / (r.std(ddof=1) + 1e-12)
        n_resamples: int = 10_000,
        seed: int,
        alternative: Alternative = "greater",
    ) -> Dict[str, float]:
        """D-05/D-06: stationary block bootstrap, one-tailed, additive-smoothed p-value."""
        if alternative != "greater":
            raise ValueError("only one-tailed 'greater' is supported (D-06)")
        d = np.asarray(paired_diffs, dtype=float)
        n = len(d)
        if n < 2:
            raise ValueError(f"paired_diffs length {n} is too short for bootstrap")
        block_size = max(2, int(math.floor(math.sqrt(n))))
        rng = np.random.default_rng(seed)
        observed = float(metric_fn(d))
        # Resample under H0: shift d so its mean is 0, then compare resampled metric to 0.
        d_centered = d - d.mean()
        count_le_zero = 0
        for _ in range(n_resamples):
            idxs = _generate_block_resample(rng, n, block_size)
            stat = float(metric_fn(d_centered[idxs]))
            if stat >= observed:
                count_le_zero += 1
        # D-06: additive-smoothed p — never reports p=0.
        p_value = (1 + count_le_zero) / (n_resamples + 1)
        return {
            "observed_metric": observed,
            "p_value": float(p_value),
            "n_oos_bars": int(n),
            "block_size": int(block_size),
            "n_resamples": int(n_resamples),
        }
    ```

    NOTE on the resample/H0 shift: D-06 says `count(stat_resampled <= 0)` — that's correct when `paired_diffs` already represents (ensemble - baseline) and the null is "metric ≤ 0". Equivalently, for a Sharpe-like metric, shift the sample so its mean is zero (centering) and then count how many of the resampled `metric_fn` values exceed the observed value. Either form is valid; the implementation above uses the centering form because it generalizes cleanly to non-mean metrics (Sharpe, dir_acc). Document this design choice as a docstring note. Tests in `<behavior>` check the outer contract (p-value behavior under known signal/noise inputs), not the centering choice.

    Tests use `pytest.mark.slow` on the n_resamples=10000 case; default `pytest -q` collects fast tests only. Add `slow` marker to `pyproject.toml` or `conftest.py` (re-use existing markers convention if present in the repo).

    ```python
    from __future__ import annotations
    from typing import Any, Dict

    P_THRESHOLD = 0.05    # D-08
    MIN_MEMBERS = 3       # CD-08

    def evaluate_win_gate(per_symbol_significance: Dict[str, Dict[str, Any]]) -> Dict[str, Any]:
        """D-08: all four conditions must hold per symbol; CD-08: < 3 members → blocked + reported."""
        out: Dict[str, Dict[str, Any]] = {}
        n_wins = 0
        for sym, sig in per_symbol_significance.items():
            reasons = []
            n_members = int(sig.get("n_members", 0))
            sharpe_p = float(sig.get("sharpe_pvalue", 1.0))
            dir_p = float(sig.get("dir_acc_pvalue", 1.0))
            sharpe_lift = float(sig.get("sharpe_lift", 0.0))
            dir_lift = float(sig.get("dir_acc_lift", 0.0))

            if n_members < MIN_MEMBERS:
                reasons.append("insufficient_runs")
            if sharpe_p >= P_THRESHOLD:
                reasons.append(f"sharpe_pvalue>={P_THRESHOLD}")
            if dir_p >= P_THRESHOLD:
                reasons.append(f"dir_acc_pvalue>={P_THRESHOLD}")
            if sharpe_lift <= 0:
                reasons.append("sharpe_lift_non_positive")
            if dir_lift <= 0:
                reasons.append("dir_acc_lift_non_positive")

            passed = len(reasons) == 0
            out[sym] = {
                **sig,
                "win_gate_passed": passed,
                "gate_failure_reasons": reasons,
            }
            if passed:
                n_wins += 1
        return {"per_symbol": out, "n_winning_symbols": n_wins}
    ```
