    ```python
    """Predict-only re-hydration of ensemble members from a snapshot row.

    Phase 4 contract (CD-11 + D-03 + Phase 3 D-08):
      - Klines reads MUST go through `load_klines_from_timescale`, which enforces
        `is_mainnet = TRUE` and the [end_ts - 365d, end_ts] window. This module
        has NO raw SQL and NO alternate klines path.
      - The OOS window end is `snapshot.config.tournament_start_ts` (CD-11).
      - Reproducibility: same snapshot row + same git_sha => identical predictions.
        Achieved by re-applying the snapshot's `seed` and re-deriving train/test
        split from the same klines slice. Phase 3 D-13 extended into Phase 4.

    This module produces the predict_fn callable consumed by
    `app.significance.predict_cache.get_or_build_predictions` on cache miss.
    """
    from __future__ import annotations
    import logging
    import re
    from datetime import datetime
    from typing import Any, Callable, Dict
    import numpy as np

    # CANONICAL klines reader — enforces is_mainnet=TRUE inside the SQL itself.
    # Do NOT reach around this; do NOT add a parallel reader (Phase 3 D-08, CLAUDE.md gotcha).
    from app.runner.data import load_klines_from_timescale

    logger = logging.getLogger(__name__)

    _SAFE_RE = re.compile(r"^[A-Za-z0-9._\-]+$")


    def _validate_id(value: str, kind: str) -> None:
        if not isinstance(value, str) or not _SAFE_RE.fullmatch(value):
            raise ValueError(f"invalid {kind}: {value!r} (must match {_SAFE_RE.pattern})")


    def _parse_iso(ts: str) -> datetime:
        # snapshot config emits ISO with timezone; tolerate "Z" suffix.
        return datetime.fromisoformat(ts.replace("Z", "+00:00"))


    def build_predict_fn(
        snapshot: Dict[str, Any],
    ) -> Callable[[Dict[str, Any]], Dict[str, np.ndarray]]:
        """Returns a callable: (row) -> {pred_prices, actual_prices, last_close}.

        The returned callable is the production predict_fn that
        `predict_cache.get_or_build_predictions` invokes on cache miss. The cache
        writes the result to `data/cache/{tournament_id}/predictions/{run_id}.npz`
        (CD-11) — this module has no I/O of its own beyond the klines read.

        Args:
            snapshot: dict matching the export_snapshot contract (config, rows, ...).

        The callable raises:
            ValueError: row.run_id or snapshot.tournament_id is path-unsafe;
                        row.status is not "success".
            AssertionError: row.symbol does not end in "USDT" (mirrors load_klines_from_timescale).
            ConnectionError: TimescaleDB unreachable (propagated from canonical loader).
        """
        tournament_id = snapshot["tournament_id"]
        _validate_id(tournament_id, "tournament_id")
        cfg = snapshot["config"]
        end_ts = _parse_iso(cfg["tournament_start_ts"])

        def _predict(row: Dict[str, Any]) -> Dict[str, np.ndarray]:
            run_id = row["run_id"]
            _validate_id(run_id, "run_id")
            if row.get("status") != "success":
                raise ValueError(
                    f"predict_fn refusing non-success row: run_id={run_id} status={row.get('status')}"
                )
            symbol = row["symbol"]
            assert symbol.endswith("USDT"), (
                f"predict_fn requires Bybit USDT-quoted symbol, got {symbol!r}"
            )
            interval = (row.get("result_json") or {}).get("interval") or row.get("interval", "5m")
            label_horizon = int(
                (row.get("result_json") or {}).get("label_horizon")
                or row.get("label_horizon", 1)
            )
            architecture = row["architecture"]
            hp_hash = row["hp_hash"]
            hyperparameters = (row.get("result_json") or {}).get("hyperparameters", {})
            seed = int(
                (row.get("result_json") or {}).get("seed")
                or cfg.get("seed", 0)
            )

            # 1. Load klines — canonical reader enforces is_mainnet=TRUE.
            df, contaminated = load_klines_from_timescale(
                symbol=symbol,
                interval=interval,
                end_ts=end_ts,
                days_back=365,
            )
            if contaminated:
                logger.warning(
                    "predict_fn: train window straddles testnet contamination cutoff for %s",
                    symbol,
                )

            # 2. Build train/test split (deterministic on `seed`).
            from app.runner.sequences import build_train_test_split  # lazy import
            split = build_train_test_split(
                df,
                label_horizon=label_horizon,
                seed=seed,
            )
            X_train = split["X_train"]
            y_train = split["y_train"]
            X_test = split["X_test"]
            y_test = split["y_test"]
            last_close_test = split["last_close_test"]

            # 3. Re-train deterministically and predict.
            from app.runner.train import build_and_train_model  # lazy — pulls TF/Keras
            model = build_and_train_model(
                architecture=architecture,
                hyperparameters=hyperparameters,
                X_train=X_train,
                y_train=y_train,
                seed=seed,
            )
            pred_prices = np.asarray(model.predict(X_test, verbose=0)).reshape(-1)
            actual_prices = np.asarray(y_test).reshape(-1)
            last_close = np.asarray(last_close_test).reshape(-1)

            n = len(pred_prices)
            if n == 0:
                raise ValueError(
                    f"predict_fn produced empty arrays for run_id={run_id} symbol={symbol}"
                )
            if actual_prices.shape != (n,) or last_close.shape != (n,):
                raise ValueError(
                    f"predict_fn shape mismatch run_id={run_id}: "
                    f"pred={pred_prices.shape} actual={actual_prices.shape} last={last_close.shape}"
                )
            return {
                "pred_prices": pred_prices,
                "actual_prices": actual_prices,
                "last_close": last_close,
            }

        return _predict
    ```
