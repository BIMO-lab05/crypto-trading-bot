// frontend/src/services/api.js — appended at end of file, alongside existing portfolioAPI/tradingAPI groups
export const tournamentAPI = {
  listSnapshots: () => api.get('/tournament/snapshots'),
  getSnapshot: (tournamentId) => api.get(`/tournament/snapshots/${tournamentId}`),
}

// frontend/src/hooks/useTournamentList.js — default export
export function useTournamentList() {
  return useQuery({
    queryKey: ['tournament-list'],
    queryFn: () => tournamentAPI.listSnapshots(),
    staleTime: Infinity,
    refetchOnWindowFocus: false,
    refetchOnMount: false,
    retry: 2,
    retryDelay: 1000,
  })
}
// returns: query.data = { success, count, tournaments: [{tournament_id, exported_at, n_rows, n_success, n_failed, architectures, symbols}, ...] }

// frontend/src/hooks/useTournamentSnapshot.js — default export
export function useTournamentSnapshot(tournamentId) {
  return useQuery({
    queryKey: ['tournament-snapshot', tournamentId],
    queryFn: () => tournamentAPI.getSnapshot(tournamentId),
    enabled: !!tournamentId,
    staleTime: Infinity,
    refetchOnWindowFocus: false,
    refetchOnMount: false,
    retry: 2,
    retryDelay: 1000,
  })
}
// returns: query.data = { success, snapshot, ensemble: object|null, significance: object|null }
